cmake_minimum_required(VERSION 3.25)

project(astra_geometry_spike
  VERSION 0.1.0
  DESCRIPTION "Conformance spike for the planned C++20 core (SO(3), SE(3), quaternion)"
  LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "build type" FORCE)
endif()

# Optimisation levels written out rather than inherited: CMake's Release default
# is -O3, while the agreed baseline is -O2.
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O0 -g" CACHE STRING "" FORCE)

# Do NOT consult (or write) the user package registry.
#
# Measured while building this spike: Eigen 3.4.0 exports itself to
# ~/.cmake/packages/Eigen3 during its configure, so a second configure reported
# "using system Eigen3 3.4.0" while no Eigen is installed on the machine --
# find_package had located the FetchContent build tree through that registry,
# silently replacing the pinned commit with a directory inside the build tree.
# Dependencies.cmake also drops a cached Eigen3_DIR; see the comment there.
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF CACHE BOOL "" FORCE)
set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON CACHE BOOL "" FORCE)

# Floating point model, fixed on purpose and recorded in the build report:
#   - no -ffast-math anywhere: it would break NaN/Inf propagation and the exact
#     structure comparisons the SE(3) last row relies on;
#   - contraction is turned OFF explicitly, so the arithmetic is the one written
#     rather than the one the compiler would prefer to fuse;
#   - no -march=native, so the binary is the same on any x86-64 host.
add_compile_options(-Wall -Wextra -Wpedantic -Werror -ffp-contract=off)

include(cmake/Dependencies.cmake)

# --- core: header-only, no JSON, no Catch2, no I/O ---------------------------
add_library(astra_core INTERFACE)
target_include_directories(astra_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(astra_core INTERFACE astra_eigen)

# --- build info, generated so a report can be fingerprinted ------------------
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/generated/astra_build_info.hpp" CONTENT
"// Generated by CMake. Do not edit.
#pragma once
#define ASTRA_COMPILER_ID \"${CMAKE_CXX_COMPILER_ID}\"
#define ASTRA_COMPILER_VERSION \"${CMAKE_CXX_COMPILER_VERSION}\"
#define ASTRA_COMPILER_PATH \"${CMAKE_CXX_COMPILER}\"
#define ASTRA_BUILD_TYPE \"${CMAKE_BUILD_TYPE}\"
#define ASTRA_CXX_STANDARD \"${CMAKE_CXX_STANDARD}\"
#define ASTRA_CXX_FLAGS \"${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${CMAKE_BUILD_TYPE}}\"
#define ASTRA_CMAKE_VERSION \"${CMAKE_VERSION}\"
#define ASTRA_EIGEN_VERSION \"${ASTRA_EIGEN_VERSION}\"
#define ASTRA_EIGEN_SOURCE \"${ASTRA_EIGEN_SOURCE}\"
#define ASTRA_EIGEN_TAG \"${ASTRA_EIGEN_GIT_TAG}\"
#define ASTRA_CATCH2_VERSION \"${ASTRA_CATCH2_VERSION}\"
#define ASTRA_CATCH2_TAG \"${ASTRA_CATCH2_GIT_TAG}\"
#define ASTRA_JSON_VERSION \"${ASTRA_JSON_VERSION}\"
#define ASTRA_JSON_TAG \"${ASTRA_JSON_GIT_TAG}\"
#define ASTRA_FFP_CONTRACT \"off\"
")

# The harness is the only layer allowed to see JSON and the generated header.
add_library(astra_harness INTERFACE)
target_include_directories(astra_harness INTERFACE
  "${CMAKE_CURRENT_SOURCE_DIR}/tests"
  "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(astra_harness INTERFACE astra_core nlohmann_json::nlohmann_json)

# --- targets -----------------------------------------------------------------
add_executable(build_info tools/build_info.cpp)
target_link_libraries(build_info PRIVATE astra_core)

# The conformance runner: consumes the Python fixtures, never modifies them.
add_executable(conformance tests/conformance.cpp)
target_link_libraries(conformance PRIVATE astra_harness)

# Native property tests and the tolerance/API-shape contracts.
add_executable(unit_tests
  tests/test_contracts.cpp
  tests/test_so3.cpp
  tests/test_quaternion.cpp
  tests/test_se3.cpp)
target_link_libraries(unit_tests PRIVATE astra_harness Catch2::Catch2WithMain)

# The allocation guards. Compiled WITHOUT NDEBUG on purpose, even in Release:
# Eigen turns eigen_assert into a no-op when NDEBUG is defined, and the Eigen
# allocation check runs through eigen_assert, so a normal Release build would arm
# that layer and detect nothing. The optimisation level is kept.
add_executable(memory_guards tests/memory_guards.cpp)
target_link_libraries(memory_guards PRIVATE astra_harness Catch2::Catch2WithMain)
target_compile_definitions(memory_guards PRIVATE EIGEN_RUNTIME_NO_MALLOC)
target_compile_options(memory_guards PRIVATE $<$<CONFIG:Release>:-UNDEBUG>)

# malloc interposition. Linux/glibc only: it relies on the dynamic linker
# resolving these symbols ahead of libc's. Where it cannot be built or used, the
# harness reports the layer as UNAVAILABLE rather than as zero.
if(UNIX AND NOT APPLE)
  add_library(interpose_malloc MODULE tools/interpose_malloc.c)
  set_target_properties(interpose_malloc PROPERTIES PREFIX "" OUTPUT_NAME "astra_interpose_malloc")
  target_link_libraries(interpose_malloc PRIVATE ${CMAKE_DL_LIBS})
endif()

# --- tests -------------------------------------------------------------------
# Dependencies.cmake held BUILD_TESTING off so that Eigen would not register its
# own suite; release that before enabling the spike's tests.
unset(BUILD_TESTING)
include(CTest)
if(BUILD_TESTING)
  # The fixtures live in the Python tree and are consumed as they are.
  set(ASTRA_FIXTURES "${CMAKE_CURRENT_SOURCE_DIR}/../benchmarks/reference")

  add_test(NAME build_info COMMAND build_info)
  add_test(NAME unit_tests COMMAND unit_tests)

  add_test(NAME conformance_schema2
    COMMAND conformance --fixtures "${ASTRA_FIXTURES}/vectors.json"
            --label schema2 --out "${CMAKE_BINARY_DIR}/conformance-schema2.json")
  add_test(NAME conformance_schema1
    COMMAND conformance --fixtures "${ASTRA_FIXTURES}/vectors-schema1.json"
            --label schema1 --out "${CMAKE_BINARY_DIR}/conformance-schema1.json")

  # The same reader, run under the preloaded allocator: this is the only way the
  # third memory layer is exercised.
  add_test(NAME memory_guards COMMAND memory_guards)
  if(TARGET interpose_malloc)
    add_test(NAME memory_guards_preloaded
      COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=$<TARGET_FILE:interpose_malloc>"
              $<TARGET_FILE:memory_guards>)
  endif()
endif()
